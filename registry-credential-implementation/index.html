    Token Generation Logic: 

    func (s *PullTokenService) GeneratePullToken(agentName, registry string) (*PullTokenResponse, error) {
    // 1. Retrieve registry credentials from store
    cred, err := s.credStore.GetCredential(registry)
    if err != nil {
        return nil, fmt.Errorf("failed to get credentials: %w", err)
    }

    // 2. Create JWT claims
    expiresAt := time.Now().Add(s.tokenTTL)
    claims := &PullTokenClaims{
        AgentName: agentName,
        Registry:  registry,
        RegisteredClaims: jwt.RegisteredClaims{
            Subject:   "pull-token",
            ExpiresAt: jwt.NewNumericDate(expiresAt),
            IssuedAt:  jwt.NewNumericDate(time.Now()),
            ID:        generateTokenID(),
        },
    }

    // 3. Sign token
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    signedToken, err := token.SignedString(s.jwtSecret)
    if err != nil {
        return nil, fmt.Errorf("failed to sign token: %w", err)
    }

    // 4. Return response with actual registry token
    return &PullTokenResponse{
        Token:     cred.Token, // Return the actual GHCR token
        Username:  cred.Username,
        Registry:  "ghcr.io",
        ExpiresIn: int(s.tokenTTL.Seconds()),
        ExpiresAt: expiresAt,
    }, nil
}

// Request/Response structures
type PullTokenRequest struct {
    Registry string `json:"registry"`
}

type PullTokenAPIResponse struct {
    Success   bool                            `json:"success"`
    Token     string                          `json:"token,omitempty"`
    Username  string                          `json:"username,omitempty"`
    Registry  string                          `json:"registry,omitempty"`
    ExpiresIn int                             `json:"expires_in,omitempty"`
    ExpiresAt string                          `json:"expires_at,omitempty"`
    Error     string                          `json:"error,omitempty"`
}

    Handler Implementation: 

    func (h *PullTokenHandlers) RequestPullTokenHandler(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
        return
    }

    // 1. Extract agent name from context (set by auth middleware)
    agentName, ok := r.Context().Value(domain.AgentNameKey).(string)
    if !ok || agentName == "" {
        http.Error(w, "Agent not authenticated", http.StatusUnauthorized)
        return
    }

    // 2. Parse request
    var req PullTokenRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "Invalid request format", http.StatusBadRequest)
        return
    }

    // 3. Validate registry type
    if req.Registry != "ghcr" {
        http.Error(w, "Unsupported registry type", http.StatusBadRequest)
        return
    }

    // 4. Generate pull token
    tokenResp, err := h.pullTokenService.GeneratePullToken(agentName, req.Registry)
    if err != nil {
        h.logger.Error("Failed to generate pull token",
            "agent", agentName,
            "registry", req.Registry,
            "error", err)
        http.Error(w, "Failed to generate pull token", http.StatusInternalServerError)
        return
    }

    // 5. Log token issuance
    h.logger.Info("Pull token issued",
        "agent", agentName,
        "registry", req.Registry,
        "expires_at", tokenResp.ExpiresAt)

    // 6. Return response
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(PullTokenAPIResponse{
        Success:   true,
        Token:     tokenResp.Token,
        Username:  tokenResp.Username,
        Registry:  tokenResp.Registry,
        ExpiresIn: tokenResp.ExpiresIn,
        ExpiresAt: tokenResp.ExpiresAt.Format(time.RFC3339),
    })
}

    Implementation: 

    func (c *PullTokenClient) RequestPullToken(registry string) (*CachedToken, error) {
    // 1. Prepare request
    reqBody := map[string]string{"registry": registry}
    jsonData, err := json.Marshal(reqBody)
    if err != nil {
        return nil, fmt.Errorf("failed to marshal request: %w", err)
    }

    // 2. Create HTTP request
    url := fmt.Sprintf("%s/api/pull-token", c.controllerURL)
    req, err := http.NewRequest("POST", url, bytes.NewBuffer(jsonData))
    if err != nil {
        return nil, fmt.Errorf("failed to create request: %w", err)
    }

    // 3. Add authentication
    req.Header.Set("Authorization", fmt.Sprintf("Bearer %s", c.agentToken))
    req.Header.Set("Content-Type", "application/json")

    // 4. Send request
    resp, err := c.client.Do(req)
    if err != nil {
        return nil, fmt.Errorf("failed to send request: %w", err)
    }
    defer resp.Body.Close()

    // 5. Parse response
    var pullTokenResp PullTokenResponse
    if err := json.NewDecoder(resp.Body).Decode(&pullTokenResp); err != nil {
        return nil, fmt.Errorf("failed to decode response: %w", err)
    }

    if !pullTokenResp.Success {
        return nil, fmt.Errorf("pull token request failed: %s", pullTokenResp.Error)
    }

    // 6. Parse expiration time
    expiresAt, err := time.Parse(time.RFC3339, pullTokenResp.ExpiresAt)
    if err != nil {
        return nil, fmt.Errorf("failed to parse expiration time: %w", err)
    }

    // 7. Cache token
    cached := &CachedToken{
        Token:     pullTokenResp.Token,
        Username:  pullTokenResp.Username,
        Registry:  pullTokenResp.Registry,
        ExpiresAt: expiresAt,
    }

    c.cacheMutex.Lock()
    c.tokenCache[registry] = cached
    c.cacheMutex.Unlock()

    c.logger.Info("Pull token received and cached",
        "registry", registry,
        "expires_at", expiresAt)

    return cached, nil
}

func (c *PullTokenClient) GetOrRequestToken(registry string) (*CachedToken, error) {
    // Check cache first
    if cached, ok := c.GetCachedToken(registry); ok {
        // Check if token expires in less than 1 minute
        if time.Until(cached.ExpiresAt) > time.Minute {
            c.logger.Debug("Using cached pull token", "registry", registry)
            return cached, nil
        }
    }

    // Request new token
    return c.RequestPullToken(registry)
}

    6. Container Runtime Integration¶

    File: internal/client/container_runtime.go (additions)

    // AuthenticateRegistry authenticates with a container registry
func (cr *ContainerRuntime) AuthenticateRegistry(registry, username, token string) error {
    cmd := exec.Command(cr.Command, "login", registry, "-u", username, "--password-stdin")
    cmd.Stdin = strings.NewReader(token)

    output, err := cmd.CombinedOutput()
    if err != nil {
        return fmt.Errorf("registry authentication failed: %w, output: %s", err, string(output))
    }

    return nil
}

// LogoutRegistry logs out from a container registry
func (cr *ContainerRuntime) LogoutRegistry(registry string) error {
    cmd := exec.Command(cr.Command, "logout", registry)

    output, err := cmd.CombinedOutput()
    if err != nil {
        return fmt.Errorf("registry logout failed: %w, output: %s", err, string(output))
    }

    return nil
}

// PullImageWithAuth pulls an image with authentication
func (cr *ContainerRuntime) PullImageWithAuth(image, registry, username, token string) error {
    // 1. Authenticate
    if err := cr.AuthenticateRegistry(registry, username, token); err != nil {
        return fmt.Errorf("authentication failed: %w", err)
    }

    // 2. Pull image
    cmd := exec.Command(cr.Command, "pull", image)
    output, err := cmd.CombinedOutput()

    // 3. Logout (always, even on error)
    defer cr.LogoutRegistry(registry)

    if err != nil {
        return fmt.Errorf("image pull failed: %w, output: %s", err, string(output))
    }

    return nil
}

    Integration Points¶

    1. Deployment Flow Integration¶

    When an agent needs to deploy a service:

    // In internal/client/reconciler.go or deployment handler

func (r *Reconciler) deployService(deployment *Deployment) error {
    // 1. Parse image from deployment
    image := parseImageFromCompose(deployment.ComposeYAML)
    registry := detectRegistry(image) // e.g., "ghcr.io"

    // 2. Request pull token if needed
    if registry == "ghcr.io" {
        token, err := r.pullTokenClient.GetOrRequestToken("ghcr")
        if err != nil {
            return fmt.Errorf("failed to get pull token: %w", err)
        }

        // 3. Pull image with authentication
        err = r.containerRuntime.PullImageWithAuth(
            image,
            token.Registry,
            token.Username,
            token.Token,
        )
        if err != nil {
            return fmt.Errorf("failed to pull image: %w", err)
        }
    }

    // 4. Continue with deployment
    return r.deployWithCompose(deployment)
}

    Security Hardening¶

    1. Encryption Key Management:
    2. Use strong random key (32+ bytes)
    3. Store in secure secret management system
    4. Rotate periodically

    5. Never commit to version control

    6. Token Security:

    7. Use HTTPS for all communication
    8. Never log token values
    9. Clear tokens from memory after use

    10. Implement rate limiting on token requests

    11. Access Control:

    12. Require admin authentication for credential management
    13. Require agent authentication for pull token requests
    14. Audit all credential operations

    15. Implement IP allowlisting if needed

    16. Monitoring:

    17. Alert on failed authentication attempts
    18. Monitor token request patterns
    19. Track credential access
    20. Log all security events
